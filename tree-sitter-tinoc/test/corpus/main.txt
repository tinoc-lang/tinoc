fn run() void {
    var sum i32 = 0;
    for 0..10 |n| {
        sum += n;
    }
    while sum > 0 {
        sum -= 1;
    }
    if sum == 0 {
        return;
    } else {
        switch sum {
            0 => { break; }
            _ => { continue; }
        }
    }
}

    Ok,
    Err(string),
}

fn divide(a f32, b f32) f32!string {
    if b == 0 {
        return Err("division by zero");
    }
    return a / b;
